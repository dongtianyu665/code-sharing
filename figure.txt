import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import matplotlib.patches as mpatches
from scipy import stats
from matplotlib.lines import Line2D
##堆叠图pgdp_sszx_percent.csv
# 创建数据
df =pd.read_csv("pgdp_sszx.csv",encoding="gbk")
#df = pd.DataFrame(data, index=years)
df
# 设置全局字体：英文使用Times New Roman，中文使用SimHei
plt.rcParams['font.family'] = ['Times New Roman', 'SimHei'] 
plt.rcParams['axes.unicode_minus'] = False

# 定义颜色（根据描述中的颜色）
pgdp_groups = sorted(df['pgdp'].unique())
colors = ['#000080', '#808000', '#008080', '#ff9999', '#808080', '#cc99cc']
treatments = ['AM', 'LM', 'HM', 'HIFU', 'VM', 'Image-guided']

# 绘制堆叠柱状图 - 单独绘制每个pgdp组的图表
for pgdp_val in pgdp_groups:
    # 创建图形和坐标轴
    fig, ax = plt.subplots(figsize=(10, 8))
    
    # 设置图形内部的背景为透明
    fig.patch.set_alpha(0.0)
    ax.patch.set_alpha(0.0)
    
    # 筛选当前pgdp组的数据
    group_data = df[df['pgdp'] == pgdp_val]
    
    # 确保年份数据正确
    years = sorted(group_data['year'].unique()) 
    
   # 为每个治疗方案的每个年份提取百分比数据
    # 创建存储每个治疗方案数据的字典
    treatment_percentages = {}
    
    for treatment in treatments:
        treatment_percentages[treatment] = []
        for year in years:
            # 从group_data中提取对应treatment和year的数据
            value = group_data[(group_data['sszx'] == treatment) & (group_data['year'] == year)]
            if not value.empty:
                treatment_percentages[treatment].append(value['percent'].iloc[0])
            else:
                # 如果没有数据，填充0
                treatment_percentages[treatment].append(0)
    
    # 初始化底部位置数组
    bottom = np.zeros(len(years))
    
    # 绘制每个治疗方案的堆叠柱状图
    for i, (treatment, color) in enumerate(zip(treatments, colors)):
        # 获取当前治疗方案的百分比数据
        values = treatment_percentages[treatment]
        
        # 绘制堆叠柱状图
        ax.bar(years, values, bottom=bottom, color=color, label=treatment, width=0.6)
        bottom += values
    
    # 设置坐标轴范围
    ax.set_ylim(0, 100)
    # 根据实际数据年份设置x轴范围
    if len(years) > 0:
        ax.set_xlim(min(years)-0.5, max(years)+0.5)
    
        #ax.set_ylim(0, 100)
        #ax.set_xlim(2015.5, 2024.5)

        # 设置Y轴刻度
        ax.set_yticks([0, 20, 40, 60, 80, 100])
        ax.set_yticks([10, 20, 30, 40, 50, 60, 70, 80, 90], minor=True)
        ax.set_yticklabels(['0%', '20%', '40%', '60%', '80%', '100%'], fontsize=50,fontweight='bold')

        # 设置X轴刻度
        ax.set_xticks(years)
        ax.set_xticklabels(years, fontsize=42,rotation=45, ha='right',fontweight='bold')

        # 移除边框，只保留坐标轴
        ax.spines['top'].set_visible(False)
        ax.spines['right'].set_visible(False)
        ax.spines['left'].set_visible(True)
        ax.spines['bottom'].set_visible(True)
        # 加粗坐标轴 - 设置线宽为2.5
        ax.spines['left'].set_linewidth(2.5)
        ax.spines['bottom'].set_linewidth(2.5)
        # 添加图例
        #ax.legend(group_data['sszx'], loc='upper right', bbox_to_anchor=(2, 1),frameon=False,prop={'weight': 'bold',"size":36})

        # 调整布局
        plt.tight_layout()

# 设置分段点
year_inter = 2019  # 可根据需要调整

##############################为每个sszx组绘制图形###################
for treatment_idx, treatment in enumerate(treatments):
    # 创建图形和坐标轴
    fig, ax = plt.subplots(figsize=(10, 8))
    
    # 设置图形内部的背景为透明
    fig.patch.set_alpha(0.0)
    ax.patch.set_alpha(0.0)
    
    # 筛选当前治疗类型的数据
    treatment_data = df[df['sszx'] == treatment]
    
    # 为每个yydj2组绘制分段拟合线
    for yydj2_idx, (yydj2_val, color) in enumerate(zip(yydj2_groups, colors)):
        # 筛选当前yydj2组的数据
        group_data = treatment_data[treatment_data['yydj2'] == yydj2_val]
        
        if not group_data.empty:
            # 按year排序
            group_data = group_data.sort_values('year1')
            
            # 提取x和y数据
            x_data = group_data['year1'].values
            y_data = group_data['zfy1'].values
            
            # 绘制散点
            ax.scatter(x_data, y_data, color=color, s=60, alpha=0.7, label=f'yydj2={yydj2_val}')
            
            # 分段线性拟合：2016年至year_inter和year_inter至2024年
            segments = [
                (min(x_data), year_inter),  # 第一段
                (year_inter, max(x_data))   # 第二段
            ]
            
            for j, (start, end) in enumerate(segments):
                # 筛选当前分段的数据
                segment_mask = (x_data >= start) & (x_data <= end)
                x_segment = x_data[segment_mask]
                y_segment = y_data[segment_mask]
                
                if len(x_segment) > 1:  # 至少需要2个点才能拟合
                    # 使用numpy的polyfit进行线性拟合
                    z = np.polyfit(x_segment, y_segment, 1)  # 1次多项式（线性拟合）
                    p = np.poly1d(z)
                    
                    # 生成拟合线的x值（从分段开始到分段结束）
                    x_fit = np.linspace(start, end, 100)
                    y_fit = p(x_fit)
                    
                    # 绘制拟合线
                    # 使用虚线表示分段拟合的连接点
                    linestyle = '-' if j == 0 or (x_fit[0] > year_inter) else '-'
                    linewidth = 2.5 if j == 0 or (x_fit[0] > year_inter) else 1.5
                    
                    # 只在第一段显示图例标签，避免重复
                    if j == 0:
                        ax.plot(x_fit, y_fit, color=color, linewidth=linewidth, 
                               linestyle=linestyle, alpha=0.9)
                    else:
                        ax.plot(x_fit, y_fit, color=color, linewidth=linewidth, 
                               linestyle=linestyle, alpha=0.9)
                    
                    # 计算置信区间
                    # 计算残差
                    residuals = y_segment - p(x_segment)
                    # 计算残差的标准差
                    residual_std = np.std(residuals)
                    # 计算标准误差
                    n = len(x_segment)
                    if n > 2:  # 需要至少3个点来计算置信区间
                        # 计算预测值的标准误差
                        x_mean = np.mean(x_segment)
                        Sxx = np.sum((x_segment - x_mean)**2)
                        # 计算每个预测点的标准误差
                        se = residual_std * np.sqrt(1/n + (x_fit - x_mean)**2 / Sxx)
                        # 计算95%置信区间 (t分布，自由度为n-2)
                        t_val = stats.t.ppf(0.975, n-2)  # 95%置信水平
                        ci = t_val * se
                        
                        # 绘制95%置信区间阴影
                        ax.fill_between(x_fit, y_fit - ci, y_fit + ci, 
                                       color=color, alpha=0.2, zorder=3)
    
    # 添加分段点垂直线，突出显示转折点
    ax.axvline(x=year_inter, color='red', linestyle='--', linewidth=2.5, alpha=0.7)

    # 根据实际数据动态设置y轴范围
    y_min = group_data['zfy1'].min() if not group_data.empty else 7000
    y_max = group_data['zfy1'].max() if not group_data.empty else 17000
    
    # 添加一些余量
    y_range = y_max - y_min
    ax.set_ylim(max(0, y_min - 0.1*y_range), y_max + 0.1*y_range)

    # 根据数据动态设置y轴刻度
    y_ticks = np.linspace(5000, 17000, 7)  # 0到10之间取6个等间隔点
    ax.set_yticks(y_ticks)
    ax.set_yticklabels([f'{int(tick/1000)}k' for tick in y_ticks], fontsize=50, fontweight='bold')
    
    # 设置X轴范围
    if not treatment_data.empty:
        years = sorted(treatment_data['year1'].unique())
        if len(years) > 0:
            ax.set_xlim(min(years)-0.5, max(years)+0.5)
            
            # 创建完整的年份范围，包括所有年份
            all_years = list(range(min(years), max(years) + 1))
            
            # 设置X轴刻度
            ax.set_xticks(all_years)
            ax.set_xticklabels(all_years, fontsize=42, rotation=45, ha='right', fontweight='bold')
    else:
        # 如果没有数据，设置默认的年份范围
        ax.set_xlim(2015.5, 2024.5)
        default_years = list(range(2016, 2025))
        ax.set_xticks(default_years)
        ax.set_xticklabels(default_years, fontsize=42, rotation=45, ha='right', fontweight='bold')

    # 添加网格线
    #ax.grid(True, alpha=0.3, linestyle='--', linewidth=0.5)
     
    # 设置坐标轴标签
    ax.set_xlabel('Year', fontsize=42, fontweight='bold', labelpad=20)
    #ax.set_ylabel('Prevalence (%)', fontsize=42, fontweight='bold', labelpad=20)
    
    # 移除边框，只保留坐标轴
    ax.spines['top'].set_visible(False)
    ax.spines['right'].set_visible(False)
    ax.spines['left'].set_visible(True)
    ax.spines['bottom'].set_visible(True)
    
    # 加粗坐标轴 - 设置线宽为2.5
    ax.spines['left'].set_linewidth(2.5)
    ax.spines['bottom'].set_linewidth(2.5)

    #             fontsize=56, fontweight='bold', pad=20)
    
    # 调整布局
    plt.tight_layout()
    
    # 显示图表
    plt.show()
